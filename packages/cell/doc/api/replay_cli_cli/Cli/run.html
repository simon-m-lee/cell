Runs the CLI with args. Returns the process exit code.

Parameters:

  • args: The command-line arguments.

Returns:

The process exit code:

  • 0 — success (verify OK, diff found no differences, fork succeeded, etc.)
  • 1 — verification failure (hash chain mismatch, divergence, diff found a difference)
  • 2 — usage error (bad arguments, missing file)
  • 3 — internal error (an unexpected exception; reported with a stack trace to stderr)

Implementation

static Future<int> run(List<String> args) async {
  if (args.isEmpty || args[0] == '--help' || args[0] == '-h') {
    stdout.writeln(_help());
    return args.isEmpty ? 2 : 0;
  }
  final command = args[0];
  final rest = args.sublist(1);
  try {
    switch (command) {
      case 'verify':
        return await _verify(rest);
      case 'diff':
        return await _diff(rest);
      case 'fork':
        return await _fork(rest);
      case 'visualise':
        return await _visualise(rest);
      case 'inspect':
        return await _inspect(rest);
      default:
        stderr.writeln('Unknown command: $command');
        stderr.writeln(_help());
        return 2;
    }
  } on _UsageException catch (e) {
    stderr.writeln('Usage error: ${e.message}');
    stderr.writeln(_help());
    return 2;
  } on FileSystemException catch (e) {
    stderr.writeln('I/O error: ${e.message} (${e.path})');
    return 2;
  } catch (e, st) {
    stderr.writeln('Internal error: $e');
    stderr.writeln(st);
    return 3;
  }
}
