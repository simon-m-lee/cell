Validates value against an explicit JsonSchema subset.

Returns null when the value fits, or a path-prefixed message when it does not. Never throws. The path form is $.target, $.address.city, $.items[0].

The enforced subset: type (integer is value is int — bool is not an integer, 1.0 is not an integer), enum, minimum / maximum (inclusive), required, properties, additionalProperties (null means closed when properties is set — schemaForValue objects are open because it sets true), and items. extra is ignored; pattern, minLength, and format are not part of this subset.

null does not satisfy a typed schema (there is no union type), and an unknown type string fails closed. The recursion depth is capped at 32 so a hostile body cannot recurse forever; past that the message is schema validation exceeded depth.

properties/required/additionalProperties are only enforced when the value is a Map. A schema that declares properties but no type: 'object' is spec-legal and leaves non-object values unchecked by those keywords — set type: 'object' when the object is meant to be closed.

Implementation

String? validateJsonSchema(
  JsonSchema schema,
  Object? value, {
  String path = r'$',
  int depth = 0,
}) {
  if (depth > 32) return 'schema validation exceeded depth';

  final type = schema.type;
  if (type != null) {
    final bool ok;
    switch (type) {
      case 'integer':
        ok = value is int;
      case 'number':
        ok = value is num;
      case 'string':
        ok = value is String;
      case 'boolean':
        ok = value is bool;
      case 'array':
        ok = value is List;
      case 'object':
        ok = value is Map;
      default:
        return '$path: unsupported type "$type"';
    }
    if (!ok) return '$path: expected $type, got ${_schemaDescribe(value)}';
  }

  final enumValues = schema.enumValues;
  if (enumValues != null) {
    final matches = enumValues.any((allowed) => _schemaEquals(value, allowed));
    if (!matches) {
      return '$path: must be one of '
          '${enumValues.map(_schemaDescribe).join(', ')}';
    }
  }

  if (value is num) {
    if (schema.minimum != null && value < schema.minimum!) {
      return '$path: below minimum ${schema.minimum}';
    }
    if (schema.maximum != null && value > schema.maximum!) {
      return '$path: above maximum ${schema.maximum}';
    }
  }

  final properties = schema.properties;
  if (value is Map) {
    for (final entry in properties?.entries ?? const <MapEntry<String, JsonSchema>>[]) {
      if (value.containsKey(entry.key)) {
        final error = validateJsonSchema(
          entry.value,
          value[entry.key],
          path: '$path.${entry.key}',
          depth: depth + 1,
        );
        if (error != null) return error;
      }
    }
    // A required key must be present; a present null is a type failure
    // on that property, not a second required failure.
    final required = schema.required;
    if (required != null) {
      for (final name in required) {
        if (!value.containsKey(name)) return '$path.$name: required';
      }
    }
    // additionalProperties: true allows unknown keys; false rejects
    // them; null means false when properties is non-null (an explicit
    // schema that models an object is closed by default).
    final additional = schema.additionalProperties;
    final closed = properties != null &&
        (additional == false || additional == null);
    if (closed) {
      for (final key in value.keys) {
        if (key is! String || !properties.containsKey(key)) {
          return '$path.$key: additional property not allowed';
        }
      }
    }
  } else if (requiredForMissingObject(schema)) {
    // `required` names a property of an object; a non-object value that
    // passed an (absent) type check still fails the requirement.
    final required = schema.required!;
    if (required.isNotEmpty) return '$path.${required.first}: required';
  }

  final items = schema.items;
  if (items != null && value is List) {
    for (var i = 0; i < value.length; i++) {
      final error = validateJsonSchema(
        items,
        value[i],
        path: '$path[$i]',
        depth: depth + 1,
      );
      if (error != null) return error;
    }
  }
  return null;
}
