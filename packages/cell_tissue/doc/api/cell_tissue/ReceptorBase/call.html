Orchestrates the formal transition from Perception to Reasoning by executing the receptor's transformation logic within the host's Operational Perimeter.

When to use

You don't call this directly – the framework invokes it when a pulse arrives at the cell. It's the core method that processes the pulse.

How it works

The method:

  1. Checks if the pulse is a shell (defensive proxy); if so, performs the mutual authorization handshake.
  2. Validates the pulse against the cell's TestCell.
  3. Executes the receptor's pipeline (preProcess → rule → postProcess).
  4. Returns the transformed pulse or null.

Non‑obvious

  • The method handles both synchronous and asynchronous execution via FutureOr.
  • If the pulse is invalidated or the cell is invalidated, the call returns null.
  • The pipeline is wrapped in error boundaries; exceptions are caught and reported via the pulse's onError callback.

Implementation

@override
FutureOr<Pulse?> call(covariant PulseBase pulse) {

  if (pulse is Shell) {
    return pulse.scrutinize(this, null);
  }

  if (!isActivated || !pulse._checker.add(cell)) return null;

  FutureOr<Pulse?> proceed() {

    if (isGoverned) {
      if (cell.isInvalidated || pulse.isInvalidated) return null;

      final ephemeralPolicy = (cell._nucleus as NucleusBase)._ephemeralPolicy;
      if (ephemeralPolicy != null) {
        ephemeralPolicy(pulse, cell: cell);
        if (pulse.isInvalidated) return null;
      }

      if (pulse.isGoverned) {

        if (pulse.isInvalidated) return null;

        final pulseEphemeralPolicy = pulse.policy;
        if (pulseEphemeralPolicy != null) {
          pulseEphemeralPolicy._onPulseComplete(pulse, cell: cell);
          if (pulse.isInvalidated) return null;
        }

        final step = cell.context is DeputyContext
            ? (cell.context as DeputyContext).role ?? cell.toString()
            : cell.toString();

        pulse = pulse.withStep(step) as PulseBase;
      }

    }

    final result = _onPulseReceived(pulse);

    if (result is Future<PulseBase?>) {
      return result.then((pulse) {
        if (pulse == null) return null;
        final nucleus = cell._nucleus as NucleusBase;
        nucleus._propagate(pulse);
        return pulse;
      });
    }

    // Synchronous Path: Immediate propagation within the current call stack.
    if (result != null) {
      final nucleus = cell._nucleus as NucleusBase;
      nucleus._propagate(result);
    }
    return result;
  }

  final validation = cell.validate(pulse, host: cell);
  if (validation is Future<bool>) {
    return validation.then((isValid) => isValid ? proceed() : null);
  } else if (!validation) {
    return null;
  }

  return proceed();

}
